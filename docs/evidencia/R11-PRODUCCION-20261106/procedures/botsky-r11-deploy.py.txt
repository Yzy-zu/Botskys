def encode_toml(s):
 text=''
 for k,v in s.items():
  if k!='calls':text+=k+' = '+json.dumps(v,ensure_ascii=False)+'\n'
 text+='\n[calls]\n';c=s['calls']
 for k,v in c.items():
  if k not in ('groups','features','schedule'):text+=k+' = '+json.dumps(v,ensure_ascii=False)+'\n'
 for group in c['groups']:
  text+='\n[[calls.groups]]\n'
  for k,v in group.items():text+=k+' = '+json.dumps(v,ensure_ascii=False)+'\n'
 for key in ('features','schedule'):
  text+='\n[calls.'+key+']\n'
  for k,v in c[key].items():text+=k+' = '+json.dumps(v,ensure_ascii=False)+'\n'
 return text
